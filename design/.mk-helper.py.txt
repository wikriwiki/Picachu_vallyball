# mk.py <md path> <node|leaf> <status>   (본문은 stdin)
# 기존 draft 의 frontmatter(id,title,parent)를 유지하고 종류별 작성 규칙 블록 + 본문으로 파일을 다시 쓴다.
import re, sys
from pathlib import Path

path = Path(sys.argv[1]); kind = sys.argv[2]; status = sys.argv[3]
body = (Path(sys.argv[4]).read_text(encoding="utf-8") if len(sys.argv) > 4 else sys.stdin.buffer.read().decode("utf-8")).strip() + "\n"
text = path.read_text(encoding="utf-8")
fm = dict(re.findall(r"^(\w+):\s*(.*)$", text.split("---")[1], re.M))
parent_rel = fm["parent"]
ptext = (path.parent / parent_rel).read_text(encoding="utf-8")
ptitle = re.search(r"^title:\s*(.*)$", ptext, re.M).group(1)
title = fm["title"]
if kind == "node":
    rules = f"""> **작성 규칙 — 이 파일을 읽거나 고치기 전에 확인**
> - 작업 전 pyramid-design 스킬의 「행동수칙」을 다시 읽는다.
> - 부모: [{ptitle}]({parent_rel}) 의 「자식 구성요소」 중 `{title}` 항목을 전개한 node 다. 부모 「통합 방식」이 이 노드에 요구하는 계약을 벗어나지 않는다. (R3)
> - 이 파일에는 자기 정의와 바로 아래 자식만 쓴다. 자식은 분해 축 하나로 2~7개, `./<id>/<id>.md` 에 둔다. (references/design-phase.md §2)
> - 「통합 방식」은 자식보다 먼저 확정한다. 자식은 이 계약을 지켜야 한다.
> - 이 파일이 바뀌면: ↑ 부모 「통합 방식」 영향 검토, ↓ 모든 자식 재검토. (references/change-protocol.md)"""
else:
    rules = f"""> **작성 규칙 — 이 파일을 읽거나 고치기 전에 확인**
> - 작업 전 pyramid-design 스킬의 「행동수칙」을 다시 읽는다.
> - 부모: [{ptitle}]({parent_rel}) 의 「자식 구성요소」 중 `{title}` 항목을 전개한 leaf 다. 부모 「통합 방식」이 요구하는 인터페이스를 그대로 구체화한다. (R3)
> - leaf 판정 기준 L1~L8 을 모두 만족해야 `designed` 가 된다. 하나라도 어려우면 node 로 바꿔 쪼갠다. (R6)
> - 구현은 이 파일의 「인터페이스」·「동작 규칙」·「경계 조건」·「테스트 케이스」만을 근거로 한다. 여기 없는 결정은 코드보다 이 파일에 먼저 적는다. (R1)
> - 이 파일이 바뀌면: 연결된 코드를 `reopen` 후 재구현하고, ↑ 부모 「통합 방식」 영향을 검토한다. (references/change-protocol.md)"""
out = f"""---
pyramid: {kind}
id: {fm['id']}
title: {title}
parent: {parent_rel}
status: {status}
---

{rules}

# {title}

{body}"""
path.write_text(out, encoding="utf-8")
print("wrote", path, kind, status)
